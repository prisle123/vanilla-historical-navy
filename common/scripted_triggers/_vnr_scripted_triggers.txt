has_naval_treaty_trigger = {
	OR = {
		has_idea = MTG_naval_treaty_adherent
        has_idea = MTG_naval_treaty_adherent_1936
		has_idea = MTG_naval_treaty_cheating
		has_idea = MTG_naval_treaty_adherent_reduced
		has_idea = MTG_naval_treaty_cheating_reduced
	}
}

expert_ai_style = {
	OR = {
		has_game_rule = { rule = ai_ship_design_style option = EXPERT }
		has_global_flag = manual_enable_expert_ship
	}
	is_in_array = { global.vnr_naval_powers = THIS }
}

expert_ai_plane = {
	OR = {
		has_game_rule = { rule = vnr_cv_air_ai option = FREE }
		has_global_flag = manual_enable_expert_plane
	}
	is_in_array = { global.vnr_naval_powers = THIS }
}

enable_vnr_naval_ai = {
	has_global_flag = vnr_enabled
	has_game_rule = { rule = vnr_ai_naval_management option = FREE }
}

is_ai_naval_major = {
	is_in_array = { global.vnr_naval_powers = THIS }
	OR = {
		is_ai = yes
		is_debug = yes
	}
}

allow_to_use_expert_navy = {
	OR = {
		AND = {
			tag = ENG
			has_game_rule = { rule = eng_start_expert_navy option = FREE }
		}
		AND = {
			tag = JAP
			has_game_rule = { rule = jap_start_expert_navy option = FREE }
		}
		AND = {
			tag = USA
			has_game_rule = { rule = usa_start_expert_navy option = FREE }
		}
	}
	OR = {
		is_ai = yes
		is_debug = yes
	}
	has_game_rule = { rule = ai_ship_design_style option = EXPERT }
}
